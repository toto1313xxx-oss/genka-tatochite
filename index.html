// 経過時間を「〇分前」「〇時間前」の形式にする関数 function formatTimeAgo(timestamp) { if (!timestamp) return '未同期'; const diffSec = Math.floor((Date.now() - timestamp) / 1000); if (diffSec < 60) return 'たった今'; const diffMin = Math.floor(diffSec / 60); if (diffMin < 60) return `${diffMin}分前`; const diffHours = Math.floor(diffMin / 60); if (diffHours < 24) return `${diffHours}時間前`; const diffDays = Math.floor(diffHours / 24); return `${diffDays}日前`; } // ヘッダーの同期表示ラベルを更新 function updateSyncDisplay() { const lastSync = localStorage.getItem('fushimidou_last_sync'); const syncEl = document.getElementById('syncStatusBadge'); if (syncEl) { if (lastSync) { syncEl.textContent = `同期: ${formatTimeAgo(Number(lastSync))}`; } else { syncEl.textContent = `同期: 未`; } } } // 1分ごとに表示（〇分前）を自動更新 setInterval(updateSyncDisplay, 60000);
